```dataviewjs
const TAG  = "あとで調べる";
const PATH = "";
const TAG_RX = new RegExp(`[＃#]${TAG}`,"u");

let rows = [];
for (const p of dv.pages(PATH)) {
  const path = p?.file?.path || "";
  if (!path.toLowerCase().endsWith(".md")) continue;

  const content = await dv.io.load(path);
  if (!content) continue;

  content.split(/\r?\n/).forEach((raw, idx) => {
    if (!TAG_RX.test(raw)) return;

    let text = raw
      .replace(/^\s*[-*]\s?/, "")
      .replace(TAG_RX, "")
      .replace(/\u00A0|\u3000/g, " ")
      .trim();
    if (!text) return;

    const ctime = p.file.ctime;               // Luxon DateTime オブジェクト
    const date  = ctime.toISODate();          // YYYY-MM-DD
    const time  = ctime.toFormat("HH:mm");    // HH:mm

    rows.push([date, time, text, p.file.link]);
  });
}

rows.length
  ? dv.table(["日付","時間","内容","ノート"], rows)
  : dv.paragraph("（該当行なし）");
```
