[[2025-09-13(土)☁️]]

select
  HPRIF.CLUB_CD as クラブコード
 ,CLUBM.CLUB_RS_NM as クラブ名
 ,SECTM.BUM_RS_NM  as 部門名
 ,case
     when HPRIF.KIND = '1' then '見学予約'
     when HPRIF.KIND = '2' then '体験予約'
     when HPRIF.KIND = '3' then '入会予約'
     when HPRIF.KIND = '4' then 'WEB入会'
     when HPRIF.KIND = '5' then '初回利用予約'  -- ★不足していた行を追加
     else ''
   end as 種別
 ,HPRIF.KAIIN_NO as 会員番号
 ,MEINF.KAIIN_NM as 会員名
 ,MEINF.KAIIN_SEI_KANA as 姓カナ
 ,MEINF.KAIIN_MEI_KANA as 名カナ
 ,case
     when HPRIF.KENGAKU_NYUKAI_STATUS = '1' then '要連絡'
     when HPRIF.KENGAKU_NYUKAI_STATUS = '2' then '予約済み'
     when HPRIF.KENGAKU_NYUKAI_STATUS = '3' then '参加済み'
     when HPRIF.KENGAKU_NYUKAI_STATUS = '4' then '取消済み'
     else ''
   end as ステータス  -- ★END を追加
 ,HPRIF.MADR as メールアドレス
 ,case
     when HPRIF.NYUKAI_RANK = '1' then '入会予約'
     when HPRIF.NYUKAI_RANK = '2' then 'フォロー中'
     when HPRIF.NYUKAI_RANK = '3' then 'フォロー開始前'
     when HPRIF.NYUKAI_RANK = '4' then 'フォロー終了'
     when HPRIF.NYUKAI_RANK = '5' then 'キャンセル待ち'
     when HPRIF.NYUKAI_RANK = '9' then 'フォロー対象外'
     else null
   end as 最終ランク
 ,HPRIF.CONTACT_YMD as 問い合わせ日
 ,HPRIF.SISETU_KENGAKU_FLG as 施設見学フラグ
 ,HPRIF.SUPPORT_STAFF_CD as フロント対応者CD
 ,HPRIF.CUSER as データ作成者
 ,EPLYM_FRONT.TAN_NM as フロント対応者
 ,HPRIF.VISITOR_WISH_YMD as 来館希望日
 ,substring(convert(char, HPRIF.VISITOR_FIX_YMD, 112), 1, 4) + '/' 
  + substring(convert(char, HPRIF.VISITOR_FIX_YMD, 112), 5, 2) + '/' 
  + substring(convert(char, HPRIF.VISITOR_FIX_YMD, 112), 7, 2) + ' '
  + substring(HPRIF.VISITOR_FIX_HM, 1, 2) + ':' + substring(HPRIF.VISITOR_FIX_HM, 3, 2)  as 来館日時
 ,LAST_MECFF.FOLLOW_NAIYO as 最終フォローコメント
 ,LAST_MECFF.MECFF_U_YMD as 最終フォロー対応日
 ,LAST_MECFF.MECFF_U_TAN_CD as 最終フォロー担当者CD
 ,EPLYM_LAST.TAN_NM as 最終フォロー担当者
 ,case
     when isnull(MEAMF_ENABLE.KAIIN_NO, '') <> '' then '有'  -- ★判定を修正
     else null
   end as 有効な契約
from HPRIF
left join SECTM
  on SECTM.BUM_CD = HPRIF.BUM_CD
left join MECFF LAST_MECFF
  on LAST_MECFF.RESERVE_NO = HPRIF.RESERVE_NO
 and LAST_MECFF.USYSDT = (
      select top 1 USYSDT
      from MECFF
      where MECFF.RESERVE_NO = LAST_MECFF.RESERVE_NO
        and MECFF.DEL_FLG <> '1'
      order by USYSDT desc
 )
left join EPLYM EPLYM_FRONT
  on EPLYM_FRONT.TAN_CD = HPRIF.SUPPORT_STAFF_CD
left join EPLYM EPLYM_LAST
  on EPLYM_LAST.TAN_CD = LAST_MECFF.MECFF_U_TAN_CD
left join MEINF
  on MEINF.KAIIN_NO = HPRIF.KAIIN_NO
left join CLUBM
  on CLUBM.CLUB_CD = HPRIF.CLUB_CD
left join (
  select
     MEAMF.KAIIN_NO
    ,HPRIF.RESERVE_NO
    ,min(MEAMF_END) as MEAMF_END
  from HPRIF
  inner join MEAMF
    on MEAMF.CLUB_CD = HPRIF.CLUB_CD
   and MEAMF.KAIIN_NO = HPRIF.KAIIN_NO
   and isnull(HPRIF.KAIIN_NO, '') <> ''
  where MEAMF.MEAMF_END >= HPRIF.COPEYMD
    and MEAMF.MEAMF_RB < 9000
  group by HPRIF.RESERVE_NO, MEAMF.KAIIN_NO
) as MEAMF_ENABLE
  on MEAMF_ENABLE.RESERVE_NO = HPRIF.RESERVE_NO
 and MEAMF_ENABLE.KAIIN_NO   = HPRIF.KAIIN_NO
 and MEAMF_ENABLE.MEAMF_END  >= HPRIF.COPEYMD
where
  HPRIF.VISITOR_WISH_YMD between '2025-08-09' and '2025-08-15'  -- 期間指定
  and HPRIF.DEL_FLG <> '1'
order by
  HPRIF.CLUB_CD
 ,HPRIF.KAIIN_NO
 ,HPRIF.KIND;
