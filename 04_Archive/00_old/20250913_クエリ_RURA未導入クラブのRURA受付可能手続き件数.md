[[20250913_Safie様依頼＿データー集計]]

対象クラブ
　富士見台3220
　稲毛2630

|   |
|---|
|DECLARE|
|@date_from date = '2025-08-30',|
|@date_to date = '2025-08-31',|
|@time_from time = '11:00:00',|
|@time_to time = '21:00:00';|
||
|WITH TARGET_CLUB AS|
|(|
|SELECT CLUB_CD, CLUB_RS_NM|
|FROM CLUBM|
|WHERE CLUB_CD IN (|
|'002630', '003220'|
|)|
|)|
||
|SELECT|
|TC.CLUB_CD AS 'クラブCD'|
|,TC.CLUB_RS_NM AS 'クラブ'|
|,CASE WHEN TC.CLUB_CD in ('001145','001150','002500','003320','005460','005480','007160') THEN 2 ELSE 3 END AS 'フェーズ'|
|,CASE WHEN (SELECT b.SYU_NM FROM MSCSM b where b.SYU_CD=M.MAE_SYU_CD) like '%おためし%' AND M.TODOKE_KBN='2' THEN 1 ELSE 0 END AS 'おた変FLG'|
|,CASE WHEN LEFT(M.CLI_NO,1) IN ('T','Z','X') THEN 1 ELSE 0 END AS 'RURA_FLG'|
|,M.SHORI_YMD|
|,M.TODOKE_KBN|
|,M.INPUT_PROCESS_MODE|
|,M.KAIIN_NO|
|,M.BUM_CD|
|,M.SYU_CD|
|,(SELECT b.SYU_NM FROM MSCSM b where b.SYU_CD=M.SYU_CD) AS 'SYU_NM'|
|,M.MAE_SYU_CD|
|,(SELECT b.SYU_NM FROM MSCSM b where b.SYU_CD=M.MAE_SYU_CD) AS 'MAE_SYU_NM'|
||
|FROM TARGET_CLUB TC|
||
|LEFT JOIN MEMRJ M|
|ON TC.CLUB_CD = M.CLUB_CD|
|AND M.SHORI_YMD BETWEEN @date_from AND @date_to|
|AND CONVERT(time, M.CSYSDT) BETWEEN @time_from AND @time_to|
|AND M.TODOKE_KBN BETWEEN 2 AND 4|
|AND NOT (M.TODOKE_KBN = 4 AND (|
|TRY_CAST(M.BUM_CD AS int) BETWEEN 2 AND 24 OR TRY_CAST(M.BUM_CD AS int) BETWEEN 103 AND 106)|
|)|
||
|ORDER BY M.KAIIN_NO|
